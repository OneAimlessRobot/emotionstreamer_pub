#include<stdlib.h>                    
#include<stdio.h>                    
#include<unistd.h>                    
#include<signal.h>                     
#include<pthread.h>                    
                    
                    
static volatile sig_atomic_t close_now=0;                    
        struct static sigaction sa;            
            
                    
static pthread_t example_tid,                    
example_tid_two;                    
                    
                    
static pthread_mutex_t exit_mtx= PTHREAD_MUTEX_INITIALIZER;                    
                    
static pthread_cond_t exit_cond= PTHREAD_COND_INITIALIZER;                    
                    
                    
                    
void close_now_or_else(void){                    
if(!close_now){  
      
pthread_mutex_lock(&exit_mtx);
close_now=1;  
pthread_cond_signal(&exit_cond);           
pthread_mutex_unlock(&exit_mtx);                
}  
                    
                    
}                    
                    
void sigint_handler_func(int useless){                    
                    
close_now=1+useless*0;                    
//be gone pedantic ass warnings              
                    
}                    
                    
                    
                    
                    
                    
void* thread_func(void* args){                    
                    
while(!close_now){                    
                    
                    
//do crap                    
}                    
                    
close_now_or_else();                    
return args;                    
                    
}                    
                    
void* thread_func_two(void* args){                    
                    
while(!close_now){                    
                    
//do some other crap i dont care about                    
}                    
                    
close_now_or_else();                    
return args;                    
                    
}                    
int main(void){                    
              
sa.sa_handler = sigint_handler_func;            
sigemptyset(&sa.sa_mask);            
sa.sa_flags = SA_RESTART;            
sigaction(SIGINT, &sa, NULL);            
            
            
pthread_create(&example_tid,NULL, &thread_func,NULL);                    
                    
pthread_create(&example_tid_two,NULL, &thread_func_two,NULL);                    
                    
                    
pthread_mutex_lock(&exit_mtx);                    
while(!close_now){                    
                    
pthread_cond_wait(&exit_cond);                    
printf("We just got told to fuck off!\nExiting!!!\n");                    
                    
}                    
pthread_mutex_unlock(&exit_mtx);                    
                    
printf("We now wait for example_tid to exit!!!\n");                    
pthread_join(example_tid,NULL);                    
printf("We just joined example_tid!!!!\n");                    
                    
                    
printf("We now wait for example_tid_two to exit!!!\n");                    
pthread_join(example_tid_two,NULL);                    
printf("We just joined example_tid_two\nExiting!!!\n");                    
                    
                    
return 0;                    
}
